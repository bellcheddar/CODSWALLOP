# The structural-compute stack, for whichever machine builds the artefacts.
#
# Split out of requirements-dev.txt because that is no longer where it lives: the droplet is
# the only worker now, so these are runtime dependencies of a production box rather than a
# workstation's toolbox. Kept separate from requirements.txt all the same, since the WEB
# process needs none of them: it reads finished JSON artefacts through readers that import
# nothing beyond the standard library, and a web worker that has never imported biotite is a
# web worker that cannot be killed by parsing a structure.
#
# All three ship prebuilt x86_64 wheels, so this needs no compiler on the droplet.
tmtools>=0.3            # TM-align bindings: the pairwise structural similarity matrix
biotite>=1.0            # mmCIF parsing, alpha-carbon extraction
gemmi>=0.6              # structure reading for the PLIP conversion
